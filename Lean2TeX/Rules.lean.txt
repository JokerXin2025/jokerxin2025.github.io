import Lean2TeX.Basic
import Mathlib.Data.Real.Basic

def Rule_Add (e : Expr) (rec : Expr → MetaM String) := do
  if e.isAppOfArity ``HAdd.hAdd 6 then
    let args := e.getAppArgs
    let A ← rec args[4]!
    let B ← rec args[5]!
    return some s! "{A}+{B}"
  return none

def Rule_Power (e : Expr) (rec : Expr → MetaM String) := do
  if e.isAppOfArity ``HAdd.hAdd 6 then
    let args := e.getAppArgs
    let A ← rec args[4]!
    let B ← rec args[5]!
    return some s! "{A}^{"{"}{B}{"}"}"
  return none

def Rule_NumSequence (e : Expr) (rec : Expr → MetaM String) := do
  if e.isApp then
    let a ← rec e.appFn!
    let n ← rec e.appArg!'
    if (← Meta.inferType e.appArg!').isConstOf ``Nat then
      return some s! "{a}{"_{"}{n}{"}"}"
  return none

def Rule_Function (e : Expr) (rec : Expr → MetaM String) := do
  if e.isApp then
    let f ← rec e.appFn!
    let x ← rec e.appArg!'
    if (← Meta.inferType e.appArg!').isConstOf ``Real then
      return some s! "{f}({x})"
  return none
